_debounce=(callback,wait){  函数防抖
  let timer;
  return ()=>{
	clearTimeOut(timer)
	timer=setTimeOut(callback,wait)
  }
}   
_throttle(callback,wait){   函数节流
  let timer;
  if(timer){
    return;
  }
  timer=setTimeOut(()=>{
   callback()
   timer=null
  },wait)
}
